is_immortal_core = {
	###The 5 Defined immortality traits only
	OR = {
		####Immortality Revised and Immortal Maldives - Even if another mod later in the load order has redifened it to be not-immortal
		has_trait = immortality_sterile
		has_trait = immortality

		has_trait = immortal
		has_trait = ageless
	}
}

is_lifetime_lock_removal_needed = {
	#Each check is wrapped in a custom_tooltip so the decision/interaction tooltip shows a readable list
	#instead of raw flag and variable names
	OR = {
		custom_tooltip = {
			text = immortality_lockout_invasion_tt
			has_character_flag = "used_lifetime_invasion"
		}
		custom_tooltip = {
			text = immortality_lockout_subjugation_tt
			has_character_flag = "used_lifetime_subjugation"
		}
		custom_tooltip = {
			text = immortality_lockout_holy_war_tt
			has_character_flag = "declared_major_religious_war_flag"
		}
		custom_tooltip = {
			text = immortality_lockout_faith_tt
			exists = var:has_created_a_faith
		}
		#Set by vanilla on_rite_edited - only the Head of Rite can edit, once per lifetime
		custom_tooltip = {
			text = immortality_lockout_rite_edit_tt
			exists = var:has_edited_a_rite
		}
		custom_tooltip = {
			text = immortality_lockout_culture_tt

		}

		custom_tooltip = {

		custom_tooltip = {

is_lifetime_vassal_lock_removal_needed = {
	OR = {
		#subject_contract_is_blocked_from_modification errors on characters without a subject contract, and tooltips evaluate every line of an AND
		#So it only ever sits behind a trigger_if limit that confirms a modifiable contract exists
		#trigger_else = { always = no } is needed because a trigger_if whose limit fails counts as true, which would make this whole OR true

		#Self - upstream contract with liege or suzerain, if present
		AND = {
			trigger_if = {
				limit = {
					is_ruler = yes
					OR = {
						is_independent_ruler = no
						is_tributary = yes
					}
					subject_contract_has_modifiable_obligations = yes
				}
				subject_contract_is_blocked_from_modification = yes
			}
			trigger_else = { always = no }
		}

		#Vassals - Downstream
		any_vassal = {
			trigger_if = {
				limit = { subject_contract_has_modifiable_obligations = yes }
				subject_contract_is_blocked_from_modification = yes
			}
			trigger_else = { always = no }
		}

		#Tributaries - Downstream
		any_tributary = {
			trigger_if = {
				limit = { subject_contract_has_modifiable_obligations = yes }
				subject_contract_is_blocked_from_modification = yes
			}
			trigger_else = { always = no }
		}
	}
}
